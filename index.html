function playAudio(text, btn) { const card = btn ? btn.closest('.card') : null; const cnChar = card ? card.querySelector('.cn-char') : null; const cnWrap = card ? card.querySelector('.cn-char-wrap') : null; if (cnChar) cnChar.classList.add('playing'); if (cnWrap) cnWrap.classList.add('playing'); if (btn) { btn.classList.add('playing'); btn.innerHTML = ''; } // التوقف عن أي نطق سابق window.speechSynthesis.cancel(); const utterance = new SpeechSynthesisUtterance(text); utterance.lang = 'zh-CN'; utterance.rate = 0.8; // جلب أصوات الجهاز وتحديد الصوت الصيني const voices = window.speechSynthesis.getVoices(); const cnVoice = voices.find(v => v.lang.includes('zh') || v.lang.includes('CN')); if (cnVoice) utterance.voice = cnVoice; utterance.onend = () => { if (btn) { btn.classList.remove('playing'); btn.innerHTML = ''; } if (cnChar) cnChar.classList.remove('playing'); if (cnWrap) cnWrap.classList.remove('playing'); }; utterance.onerror = () => { if (btn) btn.classList.remove('playing'); if (cnChar) cnChar.classList.remove('playing'); if (cnWrap) cnWrap.classList.remove('playing'); }; window.speechSynthesis.speak(utterance); if (typeof addXP === 'function') addXP(2); }
